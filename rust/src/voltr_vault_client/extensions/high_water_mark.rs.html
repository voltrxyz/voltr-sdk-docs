1//! High-water mark helpers.
2//!
3//! Mirrors `getHighWaterMarkForVault` from the JS SDK.
4
5use crate::generated::accounts::Vault;
6
7use super::decimals::decimal_bits_to_f64;
8
9/// Decoded view of a vault's [`HighWaterMark`](crate::generated::types::HighWaterMark).
10#[derive(Clone, Copy, Debug, PartialEq)]
11pub struct HighWaterMarkInfo {
12    /// Lossy `f64` view of `highest_asset_per_lp_decimal_bits`. Use the
13    /// raw `decimal_bits` field for exact comparisons.
14    pub highest_asset_per_lp: f64,
15    /// Same value as fixed-point `decimal_bits` for callers that need
16    /// exact precision.
17    pub highest_asset_per_lp_decimal_bits: u128,
18    /// Unix timestamp (seconds) at which the high-water mark was last
19    /// recalibrated.
20    pub last_updated_ts: u64,
21}
22
23/// Decodes the high-water mark fields out of `vault`.
24pub fn high_water_mark_for_vault(vault: &Vault) -> HighWaterMarkInfo {
25    HighWaterMarkInfo {
26        highest_asset_per_lp: decimal_bits_to_f64(
27            vault.high_water_mark.highest_asset_per_lp_decimal_bits,
28        ),
29        highest_asset_per_lp_decimal_bits: vault
30            .high_water_mark
31            .highest_asset_per_lp_decimal_bits,
32        last_updated_ts: vault.high_water_mark.last_updated_ts,
33    }
34}
