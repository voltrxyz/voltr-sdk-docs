3use crate::generated::accounts::Vault;
4use solana_pubkey::{pubkey, Pubkey};

6/// Number of seconds in one year, as used by the on-chain management-fee
7/// proration. Matches `ONE_YEAR_BN` in the JS SDK.
8pub const ONE_YEAR_SECONDS: u64 = 365 * 24 * 60 * 60;
9
10/// Basis-points denominator for fee math. Matches `MAX_FEE_BPS_BN` in the
11/// JS SDK.
12pub const MAX_FEE_BPS: u64 = 10_000;
13
14/// The on-chain `*_decimal_bits` fields are stored as fixed-point u128
15/// values with this many fractional bytes (i.e. `2^(8 * N)`).
16pub const DECIMAL_FRACTIONAL_BYTES: u32 = 6;
17
18/// Number of fractional bits in the fixed-point representation
19/// (`8 * DECIMAL_FRACTIONAL_BYTES`).
20pub const DECIMAL_FRACTIONAL_BITS: u32 = 8 * DECIMAL_FRACTIONAL_BYTES;
21
22/// LP withheld permanently from the very first deposit. Mirrors
23/// `programs/voltr-vault/src/utils/constants.rs::DEAD_WEIGHT`.
24pub const DEAD_WEIGHT: u64 = 1_000;

26pub const JUPITER_ADAPTOR_PROGRAM_ID: Pubkey =

28pub const DRIFT_ADAPTOR_PROGRAM_ID: Pubkey = pubkey!("EBN93eXs5fHGBABuajQqdsKRkCgaqtJa8vEFD6vKXiP");
29pub const KAMINO_ADAPTOR_PROGRAM_ID: Pubkey =
30    pubkey!("to6Eti9CsC5FGkAtqiPphvKD2hiQiLsS8zWiDBqBPKR");
31
32pub const WHITELISTED_ADAPTOR_PROGRAM_IDS: [Pubkey; 3] = [
33    JUPITER_ADAPTOR_PROGRAM_ID,
34    DRIFT_ADAPTOR_PROGRAM_ID,
35    KAMINO_ADAPTOR_PROGRAM_ID,
36];
37
38pub fn is_whitelisted_adaptor_program_id(adaptor_program: &Pubkey) -> bool {
39    WHITELISTED_ADAPTOR_PROGRAM_IDS.contains(adaptor_program)
40}
41
42pub fn vault_allows_adaptor_program(vault: &Vault, adaptor_program: &Pubkey) -> bool {
43    vault.allow_any_adaptor == 1 || is_whitelisted_adaptor_program_id(adaptor_program)
44}

123    fn detects_whitelisted_adaptors() {

130    #[test]
131    fn vault_policy_allows_whitelisted_or_bypassed_adaptors() {
132        let mut vault = empty_vault();
133        let unlisted_adaptor = Pubkey::default();
134
135        assert!(vault_allows_adaptor_program(
136            &vault,

139        assert!(!vault_allows_adaptor_program(&vault, &unlisted_adaptor));
140
141        vault.allow_any_adaptor = 1;

143        assert!(vault_allows_adaptor_program(&vault, &unlisted_adaptor));
